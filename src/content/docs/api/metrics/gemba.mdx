Scores a translation on a continuous 0-100 Scalar Quality Metric (SQM/GEMBA-DA) scale, measuring meaning preservation and grammatical correctness. Optionally calibrated using a human reference translation.

**Endpoint:** <a href="https://platform.algebras.ai/api/v1#/Translation/post_translation_gemba" target="_blank" rel="noopener noreferrer"><code>POST /translation/gemba</code></a>

## Request Body

Sent as JSON.

| Field | Type | Required | Description |
|---|---|---|---|
| `sourceLang` | string | yes | Source language code |
| `targetLang` | string | yes | Target language code |
| `sourceText` | string | yes | The original text in the source language |
| `translatedText` | string | yes | The translated text to evaluate |
| `referenceText` | string | no | Human reference translation used to calibrate the score |

<Aside type="note">
  Billed as a flat 10 words per call, regardless of text length.
</Aside>

## Response

**Success Response (200)**

```json
{
  "status": "ok",
  "timestamp": "2025-01-12T22:31:48.856Z",
  "data": {
    "score": 87.5
  }
}
```

`score` ranges from 0 (no meaning preserved) to 100 (perfect meaning and grammar).

```python
import requests

url = "https://platform.algebras.ai/api/v1/translation/gemba"
headers = {
    "X-Api-Key": "your_api_key_here",
    "Content-Type": "application/json"
}
data = {
    "sourceLang": "en",
    "targetLang": "de",
    "sourceText": "Hello, World!",
    "translatedText": "Hallo, Welt!"
}

response = requests.post(url, headers=headers, json=data)

if response.status_code == 200:
    print("Score:", response.json()["data"]["score"])
else:
    print(f"Error: {response.status_code}")
    print(response.json())
```

<TabItem label="go">
```go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
)

func main() {
    url := "https://platform.algebras.ai/api/v1/translation/gemba"

    payload := map[string]interface{}{
        "sourceLang":     "en",
        "targetLang":     "de",
        "sourceText":     "Hello, World!",
        "translatedText": "Hallo, Welt!",
    }

    jsonData, _ := json.Marshal(payload)

    req, _ := http.NewRequest("POST", url, bytes.NewBuffer(jsonData))
    req.Header.Set("X-Api-Key", "your_api_key_here")
    req.Header.Set("Content-Type", "application/json")

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        panic(err)
    }
    defer resp.Body.Close()

    body, _ := io.ReadAll(resp.Body)

    if resp.StatusCode == 200 {
        var result map[string]interface{}
        json.Unmarshal(body, &result)
        data := result["data"].(map[string]interface{})
        fmt.Println("Score:", data["score"])
    } else {
        fmt.Printf("Error: %d\n", resp.StatusCode)
        fmt.Println(string(body))
    }
}
```
